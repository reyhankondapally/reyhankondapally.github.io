#include <LiquidCrystal.h>
#include <Bonezegei_DHT11.h>

// Version 2: Temperature/humidity + smoke + infrared flame detection.
// Results are displayed on the LCD. No relay, motor, or pump is used.
#define FLAME_PIN 6
#define DHT_PIN   7
#define SMOKE_PIN 8

// LCD pins: RS, E, D4, D5, D6, D7
LiquidCrystal lcd(12, 11, 5, 4, 3, 2);
Bonezegei_DHT11 dht(DHT_PIN);

float highTemperatureF = 90.0;
const int FLAME_DETECTED_STATE = LOW;
const int SMOKE_DETECTED_STATE = LOW;
const unsigned long DHT_INTERVAL = 2000;
const unsigned long LCD_INTERVAL = 1000;

float temperatureF = 0.0;
float humidity = 0.0;
bool dhtReadingValid = false;
bool flameDetected = false;
bool smokeDetected = false;
bool highTemperature = false;

unsigned long lastDHTReadTime = 0;
unsigned long lastLCDUpdateTime = 0;

void readDHT11();
void readFlameAndSmoke();
void updateLCD();
void printLCDLine(byte row, const char *message);

void setup() {
  Serial.begin(9600);
  pinMode(FLAME_PIN, INPUT);
  pinMode(SMOKE_PIN, INPUT);

  dht.begin();
  lcd.begin(16, 2);
  printLCDLine(0, "IGNISFIRE V2");
  printLCDLine(1, "Starting...");
  delay(2000);

  readFlameAndSmoke();
  readDHT11();
  highTemperature = dhtReadingValid && (temperatureF >= highTemperatureF);
  updateLCD();

  unsigned long currentTime = millis();
  lastDHTReadTime = currentTime;
  lastLCDUpdateTime = currentTime;
}

void loop() {
  unsigned long currentTime = millis();
  readFlameAndSmoke();

  if (currentTime - lastDHTReadTime >= DHT_INTERVAL) {
    lastDHTReadTime = currentTime;
    readDHT11();
  }

  highTemperature = dhtReadingValid && (temperatureF >= highTemperatureF);

  if (currentTime - lastLCDUpdateTime >= LCD_INTERVAL) {
    lastLCDUpdateTime = currentTime;
    updateLCD();

    if (dhtReadingValid) {
      Serial.print("Temperature: ");
      Serial.print(temperatureF, 1);
      Serial.print(" F | Humidity: ");
      Serial.print(humidity, 0);
      Serial.print(" %");
    } else {
      Serial.print("DHT11: Waiting for valid reading");
    }

    Serial.print(" | Infrared flame: ");
    Serial.print(flameDetected ? "YES" : "NO");
    Serial.print(" | Smoke: ");
    Serial.println(smokeDetected ? "YES" : "NO");
  }
}

void readDHT11() {
  if (dht.getData()) {
    float newTemperatureF = dht.getTemperature(true);
    float newHumidity = dht.getHumidity();

    if (newTemperatureF >= 32.0 && newTemperatureF <= 122.0 &&
        newHumidity >= 0.0 && newHumidity <= 100.0) {
      temperatureF = newTemperatureF;
      humidity = newHumidity;
      dhtReadingValid = true;
    }
  } else {
    Serial.println("DHT11 read failed; keeping previous reading.");
  }
}

void readFlameAndSmoke() {
  flameDetected = (digitalRead(FLAME_PIN) == FLAME_DETECTED_STATE);
  smokeDetected = (digitalRead(SMOKE_PIN) == SMOKE_DETECTED_STATE);
}

void updateLCD() {
  char firstLine[17];

  if (smokeDetected && flameDetected && highTemperature) {
    printLCDLine(0, "CRITICAL FIRE!");
    printLCDLine(1, "3 SENSORS ALERT");
  } else if (smokeDetected && flameDetected) {
    printLCDLine(0, "SMOKE + FLAME");
    printLCDLine(1, "FIRE WARNING");
  } else if (smokeDetected && highTemperature) {
    printLCDLine(0, "SMOKE + HEAT");
    printLCDLine(1, "FIRE WARNING");
  } else if (flameDetected && highTemperature) {
    printLCDLine(0, "FLAME + HEAT");
    printLCDLine(1, "FIRE WARNING");
  } else if (smokeDetected) {
    printLCDLine(0, "SMOKE DETECTED");
    printLCDLine(1, "CHECK AREA");
  } else if (flameDetected) {
    printLCDLine(0, "FLAME DETECTED");
    printLCDLine(1, "CHECK AREA");
  } else if (highTemperature) {
    snprintf(firstLine, sizeof(firstLine), "HIGH TEMP %dF", (int)temperatureF);
    printLCDLine(0, firstLine);
    printLCDLine(1, "TEMP WARNING");
  } else if (!dhtReadingValid) {
    printLCDLine(0, "DHT11 WAITING");
    printLCDLine(1, "SENSORS ACTIVE");
  } else {
    snprintf(firstLine, sizeof(firstLine), "T:%dF H:%d%%",
             (int)temperatureF, (int)humidity);
    printLCDLine(0, firstLine);
    printLCDLine(1, "STATUS: SAFE");
  }
}

void printLCDLine(byte row, const char *message) {
  char line[17];
  for (int i = 0; i < 16; i++) {
    line[i] = ' ';
  }
  line[16] = '\0';

  for (int i = 0; i < 16 && message[i] != '\0'; i++) {
    line[i] = message[i];
  }

  lcd.setCursor(0, row);
  lcd.print(line);
}
